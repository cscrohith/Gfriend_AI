/// <Description>Test SIO 2 factor authentication with PaperCut auth and a web app.
///   Handle user authentication, including the initial SIO HPID linking if necessary

///   Thie will only open the app to handle/check the SIO 2 factor auth, then
///   will close the app and verify that any authentication was successful.
///   Can be used for cloud or local SIO.</Description>
/// <Precondition>SignIn_PaperCut.gfvar contains the needed authentication variables.
///   If this script is run from STF/STB the variables with names beginning
///   with STF_ can be commented to use the values from the STF/STB virtual

/// <Name>SignIn_PaperCut</Name>
/// <Description>Sign in to device with PaperCut authentication</Description>
/// <Notes>SignIn_PaperCut.gfvar contains the needed User variables.
///   If this script is run from STF/STB the variables with names beginning
///   with STF_ can be commented to use the values from the STF/STB virtual worker user.
///   This routine handles SIO as needed, including linking to an HPID account</Notes>
SignIn_PaperCut
{
    // sign in with PaperCut
    If: DeviceAuth.Sign_In_PaperCut (${STF_UserName},${UserCode},${STF_UserPassword},${Auth_TryUserID_First_YN})
    {// complete the sign in process
        DeviceAuth.CheckSignInCompletion(${HPID_UserID},${HPID_Password},${HPID_PIN})
    }
    Fail:
    {
        Fail (Failed signing in with PaperCut authentication!)
    }
    Error:
    {
        Error (Error signing in with PaperCut authentication!)
    }
}

/// <Name>OpenApp</Name>
/// <Description>Open the specified app</Description>
/// <Notes>WebApps.gfvar contains the needed app variables.</Notes>
OpenApp
{
    // open the app
    If: Device_Library.LaunchApp(${ButtonID},${ButtonText})
    {
        Sleep (${Very_Short_Wait_Time})
        // check for and handle SIO 2 factor auth PIN
        If: DeviceAuth.Check_HPID_SIO(${HPID_UserID},${HPID_Password},${HPID_PIN})
        {
            // clear out of the app and verify the the SIO 2 factor auth succeeded
            Sleep (${Very_Short_Wait_Time})
            JediOmni.Press Home Key
            JediOmni.Wait For Object (hpid-control-signin,${Wait_Time})
            
            // check for signed in linked
            If: JediOmni.Wait For Object (.hp-button.hp-button-masthead.hp-control.hp-button-text.hp-linked-cloud,${Zero_Wait_Time})
            {// the 2-factor  link succeeded
            }
            Fail:
            {// the 2-factor sign-in failed
                Fail (Failed SIO 2 Factor PIN authenticaton with PaperCut and '${ButtonText}'!)
            }
        }
        Fail:
        {
            // clear out of the app
            Sleep (${Very_Short_Wait_Time})
            JediOmni.Press Home Key
            Fail (Launch '${ButtonText}' failed!)
        }
        Error:
        {
            Error (Error occurred while launching '${ButtonText}'!)
        }
    }
    Fail:
    {
        Fail (Launch '${ButtonText}' failed!)
    }
    Error:
    {
        Error (Error occurred while launching '${ButtonText}'!)
    }
}

/// <Name>SignOut</Name>
/// <Description>Sign out on the device</Description>
/// <Notes></Notes>
SignOut
{
    DeviceAuth.SignOut
}
