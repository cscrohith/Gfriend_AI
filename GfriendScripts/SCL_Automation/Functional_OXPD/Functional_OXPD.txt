T1984767188_Validate_SafeQ_Cloud_Kiosk_Mode_OXPD_Only
{
    //Copy button
    If: JediOmni.Wait For Object (hpid-copy-homescreen-button,10)
    {
        Pass (The traditional HP home screen is displayed)   
    }
    OXPD_Common_Library.EWS_Login (${EWS_Site},${EWS_Password})
    
    //To set Kiosk mode to Secure Print
    If: Functional_OXPD.T1984767188_Validate_SafeQ_Cloud_Kiosk_Mode_SecurePrint()
    {
        JediOmni.Reset
        Oxpd.Get Browser Engine (300)
        //Secure print
        If: Oxpd.Wait For Text (Secure Print,5)
        {
            Pass (Kiosk mode has been set successfully for the Secure Print app)   
        } 
        Oxpd.Return Browser Engine
    }
    OXPD_Common_Library.EWS_Login (${EWS_Site},${EWS_Password})
    
    //To set Kiosk mode to Scan to my Email
    If: Functional_OXPD.T1984767188_Validate_SafeQ_Cloud_Kiosk_Mode_Scan_to_My_Email()
    {
        JediOmni.Reset
        Oxpd.Get Browser Engine (300)
        //Scan to my email
        If: Oxpd.Check Screen Contains Text (Scan to my email)
        {
            Pass (Kiosk mode has been set successfully for the Scan to Email app)   
        } 
        Oxpd.Return Browser Engine
    } 
}
