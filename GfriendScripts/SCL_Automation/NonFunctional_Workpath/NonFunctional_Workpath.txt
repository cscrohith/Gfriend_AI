// Provide the following details in the input variable file:

{

    Android.Touch XPath (//node[@resource-id='com.ysoft.safeq.terminals.hp.hcp:id/home_icon'])
    //Badge Authentication Stress
    Repeat:5
    {
        NonFunctional_Workpath.T1985577800_Print_Stress_Badge(${WaitTime})
    }

// Prerequisite:
// 1. Install and configure the required solution on the device.
// 2. Use the HP Card Reader Configuration Utility to configure two different cards for two different users (User1 and User2).
// 3. Register both cards to the same device/card reader.
// 4. Place both configured cards in the Badge Box before executing the test.
T1988557828_Verify_the_card_reader_behavior_when_two_cards_are_registered_to_same_device_with_different_users 
{
     // Login with User1
    BadgeBox.Scan Badge (${User1_BADGE_INDEX})

    If: JediOmni.Wait For Text (Sign Out,${WaitTime_SignIn_SignOut})
    {
        Pass (User1 logged in successfully.)
    }
    Fail:
    {
        Fail (Failed to log in User1.)
        Stop Test Case
    }

    //Swipe User2 - User1 should be logged out
    BadgeBox.Scan Badge (${User2_BADGE_INDEX})

    If: JediOmni.Wait For Text (Sign In,${WaitTime_SignIn_SignOut})
    {
        Pass (User1 logged out successfully after swiping User2's badge.)
    }
    Fail:
    {
        Fail (User1 was not logged out after swiping User2's badge.)
        Stop Test Case
    }

    //Swipe User2 again - User2 should log in
    BadgeBox.Scan Badge (${User2_BADGE_INDEX})

    If: JediOmni.Wait For Text (Sign Out,${WaitTime_SignIn_SignOut})
    {
        Pass (User2 logged in successfully.)
    }
    Fail:
    {
        Fail (Failed to log in User2.)
        Stop Test Case
    }
}
