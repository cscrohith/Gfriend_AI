// Prerequisite:
// 1. Install and configure the required solution on the device.
// 2. Configure two cards (User1 and User2) using the HP Card Reader Configuration Utility.
// 3. Register both cards to the same device.
// 4. Place both configured cards in the Badge Box.
// 5. Ensure the device is at the Sign In screen before starting the test.
T1988557798_Verify_the_card_reader_behavior_when_two_cards_are_registered_to_same_device_with_different_users 
{
  
    //Swipe User1 card - User1 Login
    BadgeBox.Scan Badge (${User1_BADGE_INDEX})

    If: JediOmni.Wait For Text (Sign Out,${WaitTime_SignIn_SignOut})
    {
        Pass (User1 logged in successfully.)
    }
    Fail:
    {
        Fail (Failed to log in User1.)
        Stop Test Case
    }

    //Swipe User2 card immediately
    BadgeBox.Scan Badge (${User2_BADGE_INDEX})

    //Verify User1 remains signed in
    If: JediOmni.Wait For Text (Sign Out,${WaitTime_SignIn_SignOut})
    {
        Pass (User1 remained signed in after User2's first swipe.)
    }
    Fail:
    {
        Fail (User1 was unexpectedly logged out after User2's first swipe.)
        Stop Test Case
    }

    //Swipe User2 card again
    BadgeBox.Scan Badge (${User2_BADGE_INDEX})

    //Verify User2 is logged in
    If: JediOmni.Wait For Text (Sign Out,${WaitTime_SignIn_SignOut})
    {
        Pass (User2 logged in successfully after the second swipe.)
    }
    Fail:
    {
        Fail (Failed to log in User2 after the second swipe.)
        Stop Test Case
    }
}
