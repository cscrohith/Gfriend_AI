\\\ Prerequisites
\\\ Configure the X3D03B_Mifare_SIM card reader using the HP Card Reader Utility Tool.
\\\ Due to Badge Box limitations, only 4 cards can be placed at a time.
\\\ Load the first set of 4 cards into the Badge Box before starting the test.
\\\ When the "Wait For User Confirm" prompt appears, replace the current cards with the next set of 4 cards and click OK to continue.
\\\ When prompted again, replace the cards with the remaining 3 cards and click OK to continue execution.
\\\ Total cards validated: 11 (executed in batches of 4 + 4 + 3).

T1984698329_Authentication_Functionality_with_HP_Universal_(Wren)_X3D03B_Mifare_SIM_(2NJ29B)_Card_Reader_and_various_Card_Types

{ 

    Repeat:4   
    {
       Calculate And Assign (${R}-1,${BadgeIndex})
       Common_Workpath_Library.Send_Jobs(${Job_Path},${Queue_Name})
       Common_Workpath_Library.Badgebox_Login(${BadgeIndex})
       Common_Workpath_Library.Pull_Print_Operations()
       Common_Workpath_Library.Device_Logout 
    }
   Wait For User Confirm (Please change other 4 set of cards)
   Repeat:4   
   {
       Calculate And Assign (${R}-1,${BadgeIndex})
       Common_Workpath_Library.Send_Jobs(${Job_Path},${Queue_Name})
       Common_Workpath_Library.Badgebox_Login(${BadgeIndex})
       Common_Workpath_Library.Pull_Print_Operations()
       Common_Workpath_Library.Device_Logout   
   }
   Wait For User Confirm (Please change other 4 set of cards)
   Repeat:3  
   {
       Calculate And Assign (${R}-1,${BadgeIndex})
       Common_Workpath_Library.Send_Jobs(${Job_Path},${Queue_Name})
       Common_Workpath_Library.Badgebox_Login(${BadgeIndex})
       Common_Workpath_Library.Pull_Print_Operations()
       Common_Workpath_Library.Device_Logout 
   }
}
