Authentication
{
    Common_Library.CheckHomeScreen
    Common_Library.AuthenticationEager(${UserName},${Password})
}

ScanToSharepointWorkFlow
{    
    //Navigate to Scan Folder
    If:Dune.Navigate To Application On Home Screen (#ScanFolderGUID)
    {
        Common_Library.SleepTime
        //clicks on Scan Folder 
        Dune.Click And Wait For Transition (#ScanFolderGUID,${ScreenRenderDelay})
        
        //wait and clicks on Scan to SharePoint app 
        Dune.Wait For (#a3d696df-b7ff-4d3d-9969-5cd7f18c0c92,5)          
        Common_Library.ClickIdWithDefaultDelay (#a3d696df-b7ff-4d3d-9969-5cd7f18c0c92 #launcherButton)        
    }
    Fail:
    {
        Return With Fail (Scan To Email app button was not found on device home screen.)
        Stop Test Suite
    }
         
    //CLICKING ON QUICKSET
    If: Dune.Wait For (${STSQuickSetName},10)
    {        
        Common_Library.ClickIdWithDefaultDelay (${STSQuickSetName})       

        Dune.Wait For (#fileNameTextField,5)        
        Common_Library.ClickIdWithDefaultDelay (#fileNameTextField)        

        Get Random Number (01,1000,${FileName})
        Common_Library.SleepTime
        Dune.Set Text (#fileNameTextField,${FileName})
        Common_Library.SleepTime
        If: Dune.Wait For (#enterKey1,2)
        {            
            Common_Library.ClickIdWithDefaultDelay (#enterKey1)
        }       
    }
    Fail:
    {
        Return With Fail (Scan to USB app is not opened)
    }    

    Common_Scan_Library.Scan_OpenOption()
}
