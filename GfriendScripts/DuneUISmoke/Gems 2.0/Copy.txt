Copy_WorkFlow
{
    //waits and clicks on the copy app in home screen
    If: Dune.Navigate To Application On Home Screen (#cedab422-33b3-4638-b6a1-604e54525215)
    {
        Dune.Click (#cedab422-33b3-4638-b6a1-604e54525215)
    }
    Fail:
    {
        Return With Fail (Copy app button was not found on the home screen)
        Stop Test Suite
    }
    
    //Options checks will starts from here
    If: Dune.Wait For (#optionsDetailPanelButton,3)
    {
        Dune.Click Id (#optionsDetailPanelButton)
    }
    Fail:
    {
        Return With Fail (unable to click on option button )
        Dune.Go To Home Screen
    }
}
