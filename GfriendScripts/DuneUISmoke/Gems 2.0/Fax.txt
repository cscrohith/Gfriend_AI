using Dune

Authentication
{
    Common_Library.CheckHomeScreen
    Common_Library.AuthenticationEager(${UserName},${Password})
}

FaxWorkFlow
{
    If:Dune.Navigate To Application On Home Screen (#44aa632b-cfa3-4c10-8cab-697a9bef610b)
    {
        Dune.Click (#44aa632b-cfa3-4c10-8cab-697a9bef610b)
    }
    Fail:
    {
        Return With Fail (Fax app button was not found on device home screen.)
        Stop Test Suite
    }
    
    //entering Send To (Fax Number)
    If: Dune.Wait For (#enterFaxNumberTextField,10)
    {
        //click on Send To Fax Number text box
        Dune.Click Id (#enterFaxNumberTextField)
            
        //types the Fax Number 
        Sleep (2)
        Dune.Fax DialPad Click(${SendTo})
    }
    Fail:
    {
        Return With Fail (Unable to update to Address)
    }
    Common_Fax_Library.Fax_OpenOption()
}
