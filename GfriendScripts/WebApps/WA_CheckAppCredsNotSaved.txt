/// <Name>Check_SIO_AppCredentials</Name>
/// <Description>Check that the credentials for the specified Web App have not
///   been saved for the authenticated device user</Description>
/// <Notes></Notes>
Check_SIO_AppCredentials
{
    // ensure any cloud connection mesage is cleared
    DeviceAuth.WaitForCloudConnection
    
    // check for the SIO Link screen
    If: JediOmni.Wait For Object (hpid-hpid-welcome-view,0)
    {
        JediOmni.Touch ID (hpid-hpid-skip-button)
        JediOmni.Wait For Text (Sign Out,10)
    }
    
    // open the app
    If: Device_Library.LaunchApp(${ButtonID},${ButtonText})
    {
    }
    Fail:
    {
        Fail (Fail launching app!)
        Stop Test Case
    }
    Error:
    {
        Error (Error launching app!)
        Stop Test Case
    }
    
    // ensure any cloud connection mesage is cleared
    DeviceAuth.WaitForCloudConnection
    
    // check for the SIO Link screen
    If: JediOmni.Wait For Object (hpid-hpid-welcome-view,0)
    {
        JediOmni.Touch ID (hpid-hpid-skip-button)
        JediOmni.Wait For Text (Sign Out,10)
    }
    
    // get the Oxpd browser
    If: Oxpd.Get Browser Engine
    {
    }
    Fail:
    {
        Fail (Fail connecting to the app OXPd!)
        Stop Test Case
    }
    Error:
    {
       Error (Error connecting to the app OXPd!)
       Stop Test Case
    }
    
    // check for EULA screen
    If: Oxpd.Wait For Css (#TermsOfUseAndPrivacyPolicy,5)
    {
    }
    Fail:
    {
        // check for popup stating leaving HP environment
        // or other pop up message
        Repeat:2
        {
            If: Oxpd.Wait For Css (#genericModel[style*="block"],5)
            {
                Oxpd.Touch Id (done-button)
                While:Oxpd.Wait For Css (#genericModel[style*="block"],1):10
                {
                }
                Sleep (10)
            }
        }
        
        // check for app main screen
        If: Oxpd.Wait For Css (#listViewHeaderContent,15)
        {
            Fail (Fail SIO user linked and app credentials saved!)
            Stop Test Case
        }
    }

    Pass (Checking that no app credentials were saved has passed!)
}
