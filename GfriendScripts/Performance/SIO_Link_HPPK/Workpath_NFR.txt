/// <Description>To measure stability of Cloud SIO app launching with 2FA for Safecom auth</Description>
/// <Author>hhgha@hp.com</Author>
/// <BlackHoleID>48</BlackHoleID>
/// <TestRailID>22486550</TestRailID>
/// <Author>hhgha@hp.com</Author>
/// <BlackHoleID>1021</BlackHoleID>
/// <TestRailID>24452633</TestRailID>
2FA_AgingWithOneDriveBusinessUsingSafecomAuth_NEW
{
    Info.Start Memory Monitoring // for checking memory usage
    Info.Start Time Check (OneDriveBSIO1stPerformance) // for checking app launching performance after SIO configuration.
   
    @Repeat:${2fa_count}
    {
       
        2fa.login_safecom_new(${2fa_id})
        
        // one drive for business
        If:JediOmni.Wait For Object (${ID_OneDriveBusiness},10)
        {
            JediOmni.Touch ID (${ID_OneDriveBusiness})
            // Input 6-pin code
            If: JediOmni.Wait For Object (hpid-hpid-pin-textbox,30)
            {
                JediOmni.Touch ID (hpid-hpid-pin-textbox)
                JediOmni.Input Text (hpid-hpid-pin-textbox,${2fa_pw})
                Sleep (2)
                JediOmni.Touch ID (hpid-hpid-signin-button)
            }
            
            Info.Mark Start(OneDriveBSIO1stPerformance,onedriveb_${R})
            If: Android.Wait For Text (Auto Test,90) // First shown folder name is "Auto Test" after Link app login automatically. 
            {
                Pass(Pass!)
                Android.Capture Screen Shot(OK_${R}.jpg)
                Info.Mark End(OneDriveBSIO1stPerformance,onedriveb_${R})
                
                Android.Press Back Key
                If:Android.Wait For Object(com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right,30)
                {
                    Android.Touch ID (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right)
                }
                
            }
            Fail:
            {
                Fail(Fail!)
                If: Android.Wait For Object (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_left,10)
                {
                    Android.Touch ID (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_left)
                }
                Android.Capture Screen Shot(NG_${R}.jpg)
                JediOmni.Capture Screen Shot(NG_${R}.jpg)
                Android.Press Back Key
                If:Android.Wait For Object(com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right,30)
                {
                    Android.Touch ID (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right)
                }
                Info.Mark End(OneDriveBSIO1stPerformance,onedriveb_${R})   
            }
        }

/// <Description>To measure reliability for SIO 1st launching with 2FA using Onedrivebusiness</Description>
/// <Author>hhgha@hp.com</Author>
/// <BlackHoleID>55</BlackHoleID>
/// <TestRailID>22486557</TestRailID>
ForOnedrivebusinessSIO1stLaunching_With2FA_Aging
{
    Info.Start Memory Monitoring
    Info.Start Time Check (OneDriveBSIO1stPerformance)
   
    Repeat:${aging_count}
    {
       
        If: Equals (${auth_method},windows)
        {
            Window_auth.WinAuthLogin(${windows_id},${windows_pw}) //Login by Window auth account.
        }
        Fail:
        {
           Sleep (1)
           If: Equals (${auth_method},hpac)
           {
               2fa.login_hpac(${2fa_id},${2fa_pw})
           }
           Fail:
           {
                Sleep (1)
                If: Equals (${auth_method},safecom)
                {
                    2fa.login_safecom(${2fa_id},${2fa_pw})
                }
                Fail:
                {
                    SecureAccess_auth(${secureaccess_id},${secureaccess_pw})
                }
           }
        } 
        
        // one drive for business
        If:JediOmni.Wait For Object (${ID_OneDriveBusiness},10)
        {
            JediOmni.Touch ID (${ID_OneDriveBusiness})
            Info.Mark Start(OneDriveBSIO1stPerformance,onedriveb_${R})
            If: Android.Wait For Text (Auto Test,90)
            {
                Info.Mark End(OneDriveBSIO1stPerformance,onedriveb_${R})
                
                Android.Press Back Key
                If:Android.Wait For Object (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right,30)
                {
                    Android.Touch ID (com.hp.print.horizontalconnector.onedriveforbusiness:id/bt_dialog_right)
                }
            }
            Fail:
            {
                Info.Mark End(OneDriveBSIO1stPerformance,onedriveb_${R})
                Android.Touch Text (OK)             
            }
        }
        
        
        If: JediOmni.Wait For Text (Sign Out,30)
        {
            JediOmni.Touch Text (Sign Out)
        }
        Info.Collect Memory Info (${R})
    }
    Info.Draw Time Graph(OneDriveBSIO1stPerformance)
   
    Info.Draw Memory Usage Graph  
}
