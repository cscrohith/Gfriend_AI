}

Authentication
{
    Pharos_Vision_AUTH.Authentication(${STF_UserName},${STF_UserPassword},${BADGE_AUTH},${STF_BadgeIndex})
}

Pharos_FLOW
{
    JediOmni.Enable Paperless Mode()

    If: Pharos_Vision_Auth.Pharos_Vision_DeletePrintSelectAll(${PRINT},${SELECT_ALL})
    { 
        //Continue  
    }
    Fail:
    {
        Return With Fail (Failed while Performing Secure print operation!)
    }
    Error:
    {
        Return With Error (Error while Performing Secure print operation!)
    } 
}

TearDown
{
    Pharos_Vision_Auth.SignOut(${STF_UserName})
}
