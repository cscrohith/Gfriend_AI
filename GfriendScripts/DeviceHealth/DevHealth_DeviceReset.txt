Using DevHealth_DeviceReset.gfvar

// when used in the STF GFriend plugin
//Using Device_EWS_Library.gflib
Using ..\Device\Device_EWS_Library.gflib

Using Web As EWS_Web

/// <Name>ResetDevice</Name>
/// <Description>Reset the device from EWS</Description>
/// <Notes></Notes>
ResetDevice
{
    // oepn the device EWS
    If: Device_EWS_Library.OpenEWS (${DUT_Address})
    {
        // sign in
        If: Device_EWS_Library.SignInWithAdmin (${DUT_AdminPW},${CheckBlankPassword},${SetPasswordWhenBlank})
        {
        }
        Fail:
        {
            Fail (Signing In with admin password failed!)
            Stop Test Case
        }
    }
    Fail:
    {
        Fail (Access device ${DUT_Address} EWS failed!)
        Stop Test Case
    }
    
    // disable web services
    If: Device_EWS_Library.DisableWebServices
    {
        // clear the proxy
        If: Device_EWS_Library.ClearProxy
        {
        }
        Fail:
        {
            Fail (Set device proxy failed!)
            Stop Test Case
        }
    }
    Fail:
    {
        Fail (Signing In with admin password failed!)
        Stop Test Case
    }

    // reset the device (partial clean from EWS)
    If: Device_EWS_Library.ResetDevice
    {
        // check for completion of the device reset
        If: Device_EWS_Library.WaitForDeviceResetComplete (${DUT_Address})
        {
        }
        Fail:
        {
            Fail (Reset device failed to complete!)
        }
    }
    Fail:
    {
        Fail (Reset device failed!)
    }
}
