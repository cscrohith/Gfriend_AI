Using ..\Common\StandardTimeouts.gfvar

using JediOmni

/// <Name>RebootDevice</Name>
/// <Description>Reboot a device</Description>
/// <Notes></Notes>
RebootDevice
{
    // ensure at the Home screen and reset
    JediOmni.Reset
    Sleep (${Very_Short_Wait_Time})
    JediOmni.Press Home Key
    Sleep (${Very_Short_Wait_Time})

    // reboot, if successful wait til Ready
    If: JediOmni.Reboot
    {
        JediOmni.Wait Until Boot
    }
    Fail:
    {
        Fail (Failed rebooting the device!)
    }
    Error:
    {
        Error (Error while rebooting the device!)
    }
}
