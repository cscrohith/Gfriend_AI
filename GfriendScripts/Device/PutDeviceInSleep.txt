Using ..\Common\StandardTimeouts.gfvar

//If the Device is HP Sercure Print or Pharos, provide the variable (${Device_is_HPSecurePrint_or_Pharos}) value as Y
//If the Device is not HP Sercure Print or Pharos, provide the variable (${Device_is_HPSecurePrint_or_Pharos}) value as N
${Device_is_HPSecurePrint_or_Pharos} = N

using JediOmni

/// <Name>PutDeviceInSleep</Name>
/// <Description>Put a device into sleep mode</Description>
/// <Notes></Notes>
PutDeviceInSleep
{
    JediOmni.Press Home Key
    
    //Click on the Direct Access Button in HP Sercure Print or Pharos devices
    If: Equals (${Device_is_HPSecurePrint_or_Pharos},Y)
    {
        Oxpd.Get Browser Engine
        Oxpd.Wait For Id (auth-button-direct-access,${Short_Wait_Time})
        Oxpd.Touch Id (auth-button-direct-access)        
    }
    // ensure the Information page element is visible
    If: JediOmni.Wait For Object (hpid-button-device-information,${Short_Wait_Time})
    {
        JediOmni.Touch ID (hpid-button-device-information)
        // click the Sleep Mode element
        JediOmni.Wait For Object (hpid-info-pages-sleep-mode-list-item,${Short_Wait_Time})
        JediOmni.Touch ID (hpid-info-pages-sleep-mode-list-item)
        // click the Sleep Now button
        JediOmni.Wait For Object (hpid-sleep-button,${Minimal_Wait_Time})
        JediOmni.Touch ID (hpid-sleep-button)
        Sleep (${Very_Short_Wait_Time})
    }
    Fail:
    {
        Fail (Failed to find the Information page link!)
    }
}
