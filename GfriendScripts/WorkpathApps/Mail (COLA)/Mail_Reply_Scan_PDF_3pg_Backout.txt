    }

        Error (Error occurred while handling the opening of '${MailButtonText}' app!)
        Stop Test Suite
    }

    If: Mail_Library.HandleSaveCredentials(Y)

    }
    Fail:

        Fail (Fail while saving app credentials '${MailButtonText}' app!)
        Stop Test Suite
    }
    Error:

        Error (Error occurred while saving app credentials '${MailButtonText}' app!)
        Stop Test Suite
    }
}

// Routine:         SendReplyJob
// Description:     send an email reply job
//
// Arguments:       
//
// Pre-requisites:  
//
// Notes:
//
// Version:
/////////////////////////////////////////////////
SendReplyJob
{
    If: Mail_Library.SearchForEmail(${Reply_Search})
    {
    }
    Fail:
    {
        Fail (Failed searching for Email ${Reply_Search}!)
        Stop Test Case
    }
    Error:
    {
        Error (Error occurred while searching for Email ${Reply_Search}!)
        Stop Test Case
    }
    
    // initiate the email reply
    If: Mail_Library.InitiateReply
    {
    }
    Fail:
    {
        Fail (Failed initiating email reply!)
        Stop Test Case
    }
    Error:
    {
        Error (Error occurred while initiating email reply!)
        Stop Test Case
    }
    
    If: Mail_Library.EnterEmailInfo(${Reply_Destination},${Reply_Subject},${Reply_Body})
    {
    }
    Fail:
    {
        Fail (Failed entering email reply info!)
        Stop Test Case
    }
    Error:
    {
        Error (Error occurred while entering email reply info!)
        Stop Test Case
    }
    
    // set the file type option
    If: Mail_Library.OpenScanOptions
    {
        If: Mail_Library.SetScanOption_FileType(${Reply_ScanFileType})
        {
            Mail_Library.CloseScanOptions
        }
        Fail:
        {
            Fail (Fail setting scan options!)
            Stop Test Case
        }
        Error:
        {
            Error (Error occurred while setting scan options!)
            Stop Test Case
        }
    }
    Fail:
    {
        Fail (Fail opening the scan options!)
        Stop Test Case
    }
    Error:
    {
        Error (Error occurred while opening the scan options!)
        Stop Test Case
    }
    
    // perform the scan
    If: Mail_Library.AddScanToEmail(${Reply_ScanFilename},${Reply_NumPages})
    {
    }
    Fail:
    {
        Fail (Fail scannig for email reply!)
        Stop Test Case
    }
    Error:
    {
        Error (Error occurred while scannig for email reply!)
        Stop Test Case
    }
    
    // send the email
    If: Mail_Library.SendEmail(${Reply_NumPages})
    {
    }
    Fail:
    {
        Fail (Fail sending email reply!)
    }
    Error:
    {
        Error (Error occurred while sending email reply!)
    }
}
