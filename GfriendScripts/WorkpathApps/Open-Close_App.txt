Using ..\Device\Device_Library.gflib

Using JediOmni

/// <Name>OpenApp</Name>
/// <Description>Open an app on a device</Description>
/// <Notes></Notes>
OpenApp
{
    If: Device_Library.LaunchApp(${ButtonID}, ${ButtonText})
    {
        Sleep (${Wait_Time})
    }
    Fail:
    {
        Fail (Fail opening the specified app!)
        Stop Test Suite
    }
    Error:
    {
        Error (Error opening the specified app!)
        Stop Test Suite
    }
}

/// <Name>CloseApp</Name>
/// <Description>Close the app on a device</Description>
/// <Notes></Notes>
CloseApp
{
    JediOmni.Press Home Key
}
