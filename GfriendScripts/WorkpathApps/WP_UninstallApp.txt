Using WP_UninstallApp.gfvar

Using WP_CommonLibrary.gflib

Using JediOmni

UninstallApp
{
    If: WP_CommonLibrary.UninstallApp(${InstallAppPath}${InstallAppFile})
    {
    }
    Fail:
    {
        Fail (Fail while uninstalling workpath app ${InstallAppFile}!)
    }
    Error:
    {
        Error (Error while uninstalling workpath app ${InstallAppFile}!)
    }
}
