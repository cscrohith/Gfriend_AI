/// <Name>SignIn</Name>
/// <Description>Sign in to HP Command Center (HPCC)</Description>
/// <Notes></Notes>
SignIn
{
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
    }
    Fail:
    {
        Fail (HPCC Sign In Failed!)
    }
    Error:
    {
        Error (Error while signing in to HPCC!)
    }
    Web.Close
}
