Using ..\Common\StandardTimeouts.gfvar

Using HPCC_Sync_Device.gfvar
Using HPCC.gfvar

Using HPCC_Library.gflib

Using JediOmni

/// <Name>VerifyDeviceSync</Name>
/// <Description>Verify the initial synch status of a device in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
VerifyDeviceSync
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    // Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        // verfiy the device status is synched
        If: HPCC_Library.VerifyDeviceSynced(${HPCC_Customer},${AssetNumber},${MaxDeviceSynchTime})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC device status verify Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC device status verify!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for device status verify Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
