//Using StandardTimeouts.gfvar
Using ..\Common\StandardTimeouts.gfvar

Using HPCC.gfvar

Using HPCC_Library.gflib

Using JediOmni

/// <Name>EnableLink</Name>
/// <Description>Enable Workpath on a device in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
EnableLink
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        //Enable Link Platform
        If: HPCC_Library.EnableLinkPlatform(${HPCC_Customer},${AssetNumber},${HPCC_TaskCompleteLimit})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (Workpath Enable Failed!)
        }
        Error:
        {
            Error (Workpath Enable Errored!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for Enable Workpath Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
