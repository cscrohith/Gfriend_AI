/// <Name>VerifyDeviceSync</Name>
/// <Description>Synch a device's info in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
DeviceSynch
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        
        Repeat:${SynchCount}
        {
            //synch the device
            If: HPCC_Library.SynchDevice(${HPCC_Customer},${AssetNumber},${MaxDeviceSynchTime})
            {
                Sleep (${Very_Short_Wait_Time})
            }
            Fail:
            {
                Fail (Device Sych Failed!)
            }
            Error:
            {
                Error (Error occured during Device Sych!)
            }
            // delay for all but the last repetition
            If: Equals (${SynchCount},${R})
            {
            }
            Fail:
            {
                Sleep (1${Long_Wait})
            }
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for Device Sych Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
