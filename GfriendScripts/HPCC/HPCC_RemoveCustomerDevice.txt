Using HPCC_Library.gflib

Using JediOmni

/// <Name>RemoveCustomerDevice</Name>
/// <Description>Remove a device in a customer
///   from HP Command Center (HPCC)</Description>
/// <Notes></Notes>
RemoveCustomerDevice
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    // Sign in
    If: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
    }
    Fail:
    {
        Fail (HPCC Sign In Failed!)
        Stop Test Case
    }
    
    // Access the customer page
    If: HPCC_Library.SelectAccount (${HPCC_Customer})
    {
        Sleep (${Short_Wait_Time})
        
        // remove device
        If: HPCC_Library.RemoveDevice_Customer(${AssetNumber})
        {
            Pass (Device removed from HPCC!)
            
            Sleep (${Short_Wait_Time})
        }
        Fail:
        {
            Fail (Fail removing device from HPCC!)
        }
        Error:
        {
            Error (Error occured while removing device from HPCC!)
        }
    }
    Fail:
    {
        Fail (Fail accessing the customer Device Discovery page!)
    }
    Error:
    {
        Error (Error accessing the customer Device Discovery page!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
