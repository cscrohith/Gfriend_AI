/// <Name>InitiateDiscoverySync</Name>
/// <Description>Initiate device discovery sync in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
InitiateDiscoverySync
{
    // Sign in
    If: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
    }
    Fail:
    {
        Fail (HPCC Sign In Failed!)
        Stop Test Case
    }
    
    // Access the customer Device Discovery page
    If: HPCC_Library.GotoCustomerDeviceDiscovery (${HPCC_Customer})
    {
        Sleep (${Short_Wait_Time})
        
        // initiate secondary device discovery sync
        If: HPCC_Library.InitiateDeviceDiscoverySync()
        {
        }
        Fail:
        {
            Fail (Fail occured during Master device discovery sync disable!)
        }
        Error:
        {
            Error (Error occured during Master device discovery sync disable!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for Master device discovery sync Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
