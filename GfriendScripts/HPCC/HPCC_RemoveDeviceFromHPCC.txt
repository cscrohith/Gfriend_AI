Using ..\Common\StandardTimeouts.gfvar

Using HPCC.gfvar

Using HPCC_Library.gflib

Using JediOmni

/// <Name>RemoveDeviceFromHPCC</Name>
/// <Description>Remove a device in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
RemoveDeviceFromHPCC
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        
        //Delete the device from unassigned devices
        If: HPCC_Library.RemoveDevice(${AssetNumber})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC Device removal Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC Device removal!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for device removal Failed!)
    }

    HPCC_Library.SignOut()
    
    Web.Close
}
