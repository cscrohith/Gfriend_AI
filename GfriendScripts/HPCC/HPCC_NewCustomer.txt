Using HPCC.gfvar

Using HPCC_Library.gflib

/// <Name>NewCustomer</Name>
/// <Description>Create a new a customer account in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
NewCustomer
{
    // Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        
        // create the new customer
        //    If: HPCC_Library.CreateCustomer(${HPCC_Customer},${HPCC_CustomerReferenceID},${HPCC_CuatomerNotes})
        If: HPCC_Library.CreateCustomer(${HPCC_Customer},,)
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC New Customer Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC New Customer!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for New Customer Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
