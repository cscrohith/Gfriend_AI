Using HPCC.gfvar

Using HPCC_Library.gflib

/// <Name>AddApp</Name>
/// <Description>Add an app to an HPCC account portfolio</Description>
/// <Notes></Notes>
AddApp
{
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        //App install
        If: HPCC_Library.AddAppToPortfolio(${App})
        {
        }
        Fail:
        {
            Fail (Add App to portfolio Failed!)
        }
        Error:
        {
            Error (Error occured during adding App to portfolio!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for adding App to portfolio Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
