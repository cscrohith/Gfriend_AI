Using HPCC.gfvar

Using HPCC_Library.gflib

Using JediOmni

/// <Name>UnassignAndRemoveDevice</Name>
/// <Description>Unassign a device from a customer account in the device list
///   then remove it from HP Command Center (HPCC)</Description>
/// <Notes></Notes>
UnassignAndRemoveDevice
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    // Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        
        // Unassign the device from the account
        If: HPCC_Library.UnassignDevice(${AssetNumber})
        {
            Sleep (${Very_Short_Wait_Time})
        
            //Delete the device from unassigned devices
            If: HPCC_Library.RemoveDevice(${AssetNumber})
            {
                Sleep (${Very_Short_Wait_Time})
            }
            Fail:
            {
                Fail (Fail occured during HPCC Device removal!)
            }
            Error:
            {
                Error (Error occured during HPCC Device removal!)
            }
        }
        Fail:
        {
            Fail (Fail occurred during HPCC Device unassigning!)
        }
        Error:
        {
            Error (Error occured during HPCC Device unassigning!)
        }
    }
    Fail:
    {
        Fail (Fail occurred while signing in to HPCC for device unassing and remove!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
