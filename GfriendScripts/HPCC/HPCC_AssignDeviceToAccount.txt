Using ..\Common\StandardTimeouts.gfvar

Using HPCC.gfvar

Using HPCC_Library.gflib

Using Info
Using JediOmni

/// <Name>AssignDeviceToAccount</Name>
/// <Description>Assign an unassigned device to a customer account</Description>
/// <Notes></Notes>
AssignDeviceToAccount
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    //Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        //Unassign the device from the account
        If: HPCC_Library.AssignDevice(${HPCC_Customer},${AssetNumber})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC Device assigning Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC Device assigning!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for device assigning Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
