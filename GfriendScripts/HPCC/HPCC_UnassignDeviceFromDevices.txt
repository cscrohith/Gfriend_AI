Using HPCC_Library.gflib

Using JediOmni

/// <Name>UnassignDeviceFromDevices</Name>
/// <Description>Unassign a device from a customer account in the device list
///   in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
UnassignDeviceFromDevices
{
    // remove the "-" from the device as it is a non-allowed character in HPCC
    Set Variable (${AssetNumber},${DUT_ID})
    Replace (${AssetNumber},-,)
    
    // Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        
        // Unassign the device from the account
        If: HPCC_Library.UnassignDevice(${AssetNumber})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC Device unassigning Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC Device unassigning!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for device unassigning Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
