//Using StandardTimeouts.gfvar
Using ..\Common\StandardTimeouts.gfvar

Using HPCC.gfvar

Using HPCC_Library.gflib

/// <Name>RemoveCustomer</Name>
/// <Description>Delete a customer account in HP Command Center (HPCC)</Description>
/// <Notes></Notes>
RemoveCustomer
{
    // Signin
    IF: HPCC_Library.SignIn (${HPCC_User},${HPCC_PassWord},${HPCC_url},${HPCC_Browser})
    {
        Sleep (${Very_Short_Wait_Time})
        // delete the customer
        If: HPCC_Library.DeleteCustomer(${HPCC_Customer})
        {
            Sleep (${Very_Short_Wait_Time})
        }
        Fail:
        {
            Fail (HPCC Remove Customer Failed!)
        }
        Error:
        {
            Error (Error occured during HPCC Remove Customer!)
        }
    }
    Fail:
    {
        Fail (HPCC Sign In for Remove Customer Failed!)
    }
    
    HPCC_Library.SignOut()
    
    Web.Close
}
